#include<stdio.h>
int main()
{
    int course;
    float credit, gpa, sum=0, sum_credit=0,cgpa; 
    printf("Number of courses : ");
    scanf("%d",&course);

    for(int i=0; i<course; i++)
    {
        printf("Credit of course[%d] : ",i+1 );
        scanf("%f",&credit);

        printf("Earned GPA of course[%d] : ", i+1);
        scanf("%f",&gpa);

        sum=sum+credit*gpa;
        sum_credit=sum_credit+credit;
    }

    printf("\n");
    cgpa=sum/sum_credit;
    printf("CGPA : %.2f", cgpa);
    printf("\n");

    if(cgpa<=4.00 && cgpa>=3.50)
    {
        printf("Outstanding/First Class");
    }
    else if(cgpa<=3.49 && cgpa>=3.00)
    {
        printf("Very Good/First Class");
    }
    else if(cgpa<=2.99 && cgpa>=2.50)
    {
        printf("Good/Second Class");
    }
    else if(cgpa<=2.49 && cgpa>=2.49)
    {
        printf("Average/pass");
    }
    else
    {
        printf("Fail");
    }
}
